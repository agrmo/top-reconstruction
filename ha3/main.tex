\section*{Problem 1}
For reference,
\e{f(x)\odd \times g(x)\odd=\even}
\e{f(x)\odd \times g(x)\even=\odd}
\e{f(x)\even \times g(x)\even=\even}
Find $a_0$, $a_k$, $b_k$.
\e{L=2p\quad \omega=\fr{2\pi}{L}=\fr{\pi}{p}}
\e{a_0=\fr{1}{2p} \int_{-p}^{p} f(x) dx = \fr{1}{2p} \int_{-p}^{p} \odd dx = \boxed{0}}
\e{a_k=\fr{1}{p} \int_{-p}^{p} f(x) \cos(k\omega x) dx = \int\odd\even = \int\odd = \boxed{0}}
\e{b_k=\fr{1}{p} \int_{-p}^{p} f(x) \sin(k\omega x) dx &= \fr{2}{p} \int_{0}^{p} \sin(\pi k p^{-1}) dx\\
&=\fr{2p^{-1}p}{\pi k} (\cos(0)-\cos(\pi k))\\
&=\fr{2}{\pi k}(1-\cos(\pi k))=\fr{2}{\pi k}(1-(-1)^k)\\
&=\boxed{\begin{cases}0&\textrm{k even}\\ \fr{4}{\pi k}&\textrm{k odd}\end{cases}}}
\e{\boxed{f(x)=\sum_{k=0}^{\infty} \fr{4}{\pi(2k+1)} \sin(\fr{(2k+1)}{p}x)}}

\section*{Problem 8}
\e{r\in \mathbb{R}}
\e{y=x^r}
\e{x^2y''\textrm{ and }xy'=\textrm{ scalar multiple of }y}
\e{x^2y''+axy' + by =0}
\e{r(r-1)x^r+arx^r + bx^r=0}
\e{x^r(r(r-1)+ar+b)=0}
\e{r(r-1)+ar+b=0}
\e{r^2+(a-1)r+b=0}
The roots of the Euler equation are of the form
\e{\boxed{r=\fr{1}{2}(1-a\pm((1-a)^2-4b))^{1/2}}}
The solutions can be found in cases.
\subsection*{Case $(a-1)^2-4b>0.$}
Use the case for 2nd order ODEs with constant coefficients where the discriminant is positive. Two roots $r_1\neq r_2$ and $r_1,r_2\in \mathbb{R}$.
\e{r_1=\fr{1}{2}(1-a+((1-a)^2-4b)^{1/2})}
\e{r_2=\fr{1}{2}(1-a-((1-a)^2-4b)^{1/2})}
The general solution is
\e{y(x)&=c_1 e^{r_1 t}+c_2 e^{r_2 t}\\
&=c_1 e^{r_1 \ln x}+c_2 e^{r_2 \ln x}\\
&=c_1 e^{\ln x^{r_1}}+c_2 e^{\ln x^{r_2}}\\
&=\boxed{c_1 x^{r_1}+c_2 x^{r_2}}}
We derived that the solutions are in the form
\e{\boxed{y=x^r}}
and linearly independent. Use this solution for the rest of the cases.
\subsection*{Case $(a-1)^2-4b=0.$}
Use the case for 2nd order ODEs with constant coefficients where the discriminant is 0. The root is repeated.
\e{r=\fr{1}{2}(1-a)}
We need two linearly independent solutions for any second order ODE. The other solution can be derived by viewing the characteristic equation and referring to the second order ODE solution for $y(t)$ in this case.
\e{y''+ay'+by=0}
\e{\lambda^2+a\lambda+b=0}
\e{y(t)=c_1 e^{\lambda t} + c_2te^{\lambda t}}
Then the general solution $y$ is 
\e{\Aboxed{y(x)&=c_1 x^r+c_2 x^r \ln(x)}}
\subsection*{Case $(a-1)^2-4b<0.$}
Use the case for 2nd order ODEs with constant coefficients where the discriminant is negative.
\e{r=r_0\pm s_0 i = \fr{1}{2}(1-a)\pm \fr{1}{2}(4b-(1-a)^2)^{1/2}i}
\e{x^r &= x^{r_0 \pm s_0 i}=x^{r_0}x^{s_0 i}=x^{r_0}e^{\ln (x)s_0 i}\\
&= x^{r_0}(\cos(s_0 \ln(x))+i\sin(s_0 \ln(x)))}
The real part is a solution, and the imaginary part is a solution, and they are linearly independent. The general solution $y$ is a linear combination of these solutions.
\e{\Aboxed{y(x)&=c_1 x^{r_0}\cos(s_0 \ln(x)) + c_2 x^{r_0} \sin(s_0 \ln(x))}}
