\section*{Problem 2}
$J_n(x)$ and $Y_n(x)$ are not scalar multiples of eachother. $x^n J_n(x)$ and $x^n Y_n(x)$ are not scalar multiples of eachother, so they are \fbox{lienarly independent}. Check that $y=x^n J_n(x)$ solves the Bessel equation.
\e{\fr{d}{dx} (x^n J_n(x)) = x^n J_n' + nx^{n-1} J_n}
\e{\fr{d^2}{dx^2} (x^n J_n(x)) = x^n J_n'' +n x^{n-1} J_n' + n x^{n-1}J_n' + n(n-1)x^{n-2}J_n}
Plug into the ODE,
\e{\label{nsc8}xy'' + (1-2n)y' + xy &=0\\
x^{n+1}J_n'' nx^n J_n' + n x^{n} J_n' + n(n-1)x^{n-1} J_n + (1-2n)(x^n J_n' + n x^{n-1} J_n) + x^{n+1} J_n &= 0\\
x^{n-1}(x^2 J_n'' + xnJ_n' + xnJ_n' + n^2 J_n - nJ_n + xJ_n' + nJ_n - 2nxJ_n' - 2n^2J_n + xJ_n)&=0\\
\label{2n34}x^{n-1}(x^2 J_n'' + xJ_n' + (x-n^2)J_n)&=0
\intertext{The term in parenthesis is a Bessel equation with $y(x)=J_n(x)$ and $\mu=1$. Because $J_n(x)$ solves the Bessel equation, the term in parenthesis is equal to 0.}
x^{n-1}(0)&=0\\
0&=0}
So \fbox{$y=x^n J_n(x)$ solves the ODE \eref{nsc8}}. Because $Y_n(x)$ also solves the Bessel equation, the same procedure will result in \eref{2n34}, so \fbox{$y=x^n Y_n(x)$ solves the ODE \eref{nsc8}}.

\section*{Problem 4}
\e{y=x^{-n}U}
\e{y'=-nx^{-n-1}U+x^{-n}U'}
\e{y''=-n(-n-1)x^{-n-2}U-n^{-n-1}U'-nx^{-n-1}U'+x^{-n}U''}
\e{xy''=n(n+1)x^{-n-1}U-nx^{-n}U'-nx^{-n}U'+x^{-n+1}U''}
\e{(1+2n)y'=-n(1+2n)x^{-n-1}U+(1+2n)x^{-n}U'}
\e{xy=x^{-n+1}U}
These are the terms of the not-Bessel ODE. Plug them in to find the Bessel equation for $U$.
\e{xy''+(1+2n)y'+xy&=0\\
n(n+1)x^{-n-1}U-nx^{-n}U'-nx^{-n}U'+x^{-n+1}U''-n(1+2n)x^{-n-1}U+(1+2n)x^{-n}U'+x^{-n+1}U&=0\\
n(n+1)x^{-1}U-nU'-nU'+xU''-n(1+2n)x^{-1}U+(1+2n)U'+xU&=0\\
n(n+1)U-2nxU'+x^2U''-n(1+2n)U+(1+2n)U+(1+2n)xU'+x^2U&=0\\
x^2U''(n(n+1)-(-2n-1+2n)xU'-n(1+2n)+x^2)U&=0\\
\Aboxed{x^2U''+xU'+(x^2-n^2)U&=0}}
The non-Bessel ODE for $y$ is in the form of the Bessel equation for $U$ with $\boxed{\mu^2=1}$. We can covert the solution for $U$ into a solution for $y$. The general solution for $U$ is
\e{\boxed{U(x)=c_1 J_n(x)+c_2 Y_n(x)}}
which using $y=x^{-n}U$ is the general solution for $y$,
\e{\boxed{y(x)=c_1 x^{-n} J_n(x) + c_2 x^{-n} Y_n(x)}}

\e{a=1\quad \mu_j=\fr{\alpha_j}{a}=\alpha_j \quad n=0}
\e{A_j &= \fr{2}{(1)^2 J_1^2(\alpha_j)}\int_0^c (1)J_0(\alpha_j x)x dx

\e{f(x)&=\sum_{j=1}^\infty A_J J_0(\mu_j x)\\
