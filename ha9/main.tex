\section*{Problem 1}
This is a two dimensional wave equation with vanishing Dirichlet boundary conditions. Find $A(n_1,n_2)$, $B(n_1,n_2)$, then the solution $u(x,y)$.
\e{l_1=1\quad l_2=1 \quad c=\fr{1}{\pi}}
\e{\lambda=-(n_1^2+n_2^2)\pi^2}
\e{A=4\int_0^1 \int_0^1 (0) dx dy = \boxed{0}}
\e{B&=4\pi|\lambda|^{-1/2}\int_0^1 \int_0^1 \sin(\pi n_1 x)\sin(\pi n_2 y) dxdy\\
&=\fr{4 \pi |\lambda|^{-1/2}}{\pi^2 n_1 n_2}((-1)^{n_1}-1)((-1)^{n_2}-1)\\
&=\boxed{\falls{\fr{16}{\pi^2 n_1 n_2 (n_1^2+n_2^2)^{1/2}} &\quad \textrm{$n_1$ odd and $n_2$ odd}\\
0 &\quad \textrm{$n_1$ even or $n_2$ even}}}}
The solution is
\e{
u(t,x,y)&=\sum_{n_1=1}^\infty \sum_{n_2=1}^\infty B(n_1,n_2)\sin(|\lambda|^{1/2}ct)\sin(\pi n_1 x) \sin(\pi n_2 y)\\
&= \sum_{k_1=0}^\infty \sum_{k_2=0}^\infty \fr{16}{\pi^2 (2k_1+1)(2k_2+1)((2k_1+1)^2+(2k_2+1)^2)^{1/2}}
\\ &\quad + \sin(\pi((2k_1+1)^2+(2k_2+1)^2)t)\sin(\pi(2k_1+1)x)\sin(\pi(2k_2+1)y)
}
\section*{Problem 2}
This is a two dimensional wave equation with vanishing Dirichlet boundary conditions. Find $A(n_1,n_2)$, $B(n_1,n_2)$, then the solution $u(x,y)$.
\e{l_1 = 1 \quad l_2=1}
\e{\lambda=-(n_1^2+n_2^2)\pi^2}
\e{A(n_1,n_2)&=4\int_0^1\int_0^1 x(1-x)y(1-y)\sin(\pi n_1 x)\sin(\pi n_2 y)dx dy\\
&= \boxed{\falls{\fr{64}{\pi^6 n_1^3 n_2^3} &\quad \textrm{$n_1$ odd and $n_2$ odd}\\
0 &\quad \textrm{$n_1$ even or $n_2$ even}}}}
\e{B(n_1,n_2)&=4\pi|\lambda|^{-1/2}\int_0^1 \int_0^1 2\sin(\pi x)\sin(2\pi y)\sin(\pi n_1 x)\sin(\pi n_2 y)dx dy
\intertext{Use the orthogonality of sine for the two integrals.}
&= \falls{8\pi|\lambda|^{-1/2}(\fr{1}{2})(\fr{1}{2}) &\quad \textrm{$n_1=1$ and $n_2=2$}\\
0 &\quad \textrm{$n_1\neq 1$ or $n_2\neq 2$}}\\
&= \boxed{\falls{\fr{2}{\sqrt{5}} &\quad  \textrm{$n_1=1$ and $n_2=2$}\\
0 &\quad \textrm{$n_1\neq 1$ or $n_2\neq 2$}}}}
\e{\lambda(n_1=1,n_2=2)=-(1^2+2^2)\pi^2=-5\pi^2}
The solution is
\e{u&=\sum_{n_1=1}^\infty \sum_{n_2=1}^\infty (A(n_1,n_2)\cos(|\lambda|^{1/2}\pi^{-1} t) + B(n_1,n_2)\sin(|\lambda|^{1/2} \pi^{-1} t))\sin(\pi n_1 x)\sin(\pi n_2 y)
\intertext{There is only one case for $B$, and $A$ is nonzero only when $n_1$ and $n_2$ are odd. The $\pi$ from $\lambda$ cancels with the $\pi$ from $ct$.}
&= \fr{2}{\sqrt{5}}\sin(\sqrt{5} t)\sin(\pi x)\sin(2\pi y)\\
&\quad + \sum_{k_1=0}^\infty \sum_{k_2=0}^\infty \fr{64}{\pi^6 (2k_1+1)^3 (2k_2+1)^3} \cos( ((2k_1+1)^2+(2k_2+1)^2)^{1/2} t)\sin(\pi (2k_1+1)x)\sin(\pi (2k_2+1) y)}
\section*{Problem 3}
This is a two dimensional Laplace equation with mixed Neumann and Dirichlet boundary conditions. Find the coefficients $A_n$, $B_n$, then $u(x,y)$ by applying the box boundary conditions.
\e{u(x,y)=\sum_{n=1}^\infty (A_n \cosh(\pi l_1^{-1} ny)+B_n \sinh (\pi l_1^{-1} n y)) \sin(\pi l_1^{-1} n x)}
\e{u(x,0)=0\Rightarrow u(x,0)=\sum_{n=1}^\infty A_n \sin(\pi l_1^{-1} x) \Rightarrow \boxed{A_n = 0}}
The solution is $u(x,y)$.
\e{\boxed{u(x,y)=\sum_{n=1}^\infty B_n \sinh (\pi l_1^{-1} n y) \sin(\pi l_1^{-1} n x)}}
\e{\fr{\p}{\p y} u(x,l_2) = \sum_{n=1}^\infty B_n \pi l_1^{-1} n \cosh(\pi l_1^{-1} n l_2) \sin(\pi l_1^{-1} n x)=f(x)}
This is the Fourier series for $f(x)$. The coefficients of the Fourier series are $B_n$.
\e{\boxed{B_n = \fr{2}{\pi n \cosh(\pi l_1^{-1} l_2 n)} \int_0^{l_1} f(x) \sin(\pi l_1^{-1} n x)dx}}
\section*{Problem 4}
This is an application of Problem 3. This is a two dimensional Laplace equation with mixed Neumann and Dirichlet boundary conditions. Find the coefficients $A_n$, $B_n$, then $u(x,y)$ by applying the box boundary conditions.
\e{u(x,y)=\sum_{n=1}^\infty B_n \sinh (\pi l_1^{-1} n y) \sin(\pi l_1^{-1} n x)}
Find $B_n$.
\e{B_n &= \fr{2}{\pi n \cosh(\pi l_1^{-1} l_2 n)} \int_0^{l_1} f(x) \sin(\pi l_1^{-1} n x)dx\\
&= \fr{2}{\pi n \cosh(\pi n)} \int_0^{l_1} f(x) \sin(\pi l_1^{-1} n x) dx\\
&= \boxed{\falls{\fr{1}{2 \cosh(2 \pi)} &\quad n=2\\
0 &\quad n\neq 2}}}
The solution is $u(x,y)$.
\e{\boxed{u(x,y)=\fr{1}{2 \cosh(2 \pi)} \sin(2x)\sin(2y)}}
\section*{Problem 5}
This is a two dimensional Laplace equation with inhomogenous boundary conditions. The solution is
\e{u(x,y)=\sum_{n=1}^\infty (A_n \cosh(\pi l_1^{-1} n (\beta - y)) + B_n \sinh (\pi l_1^{-1} (\beta-y)))\sin(\pi l_1^{-1} n x)}
Solve for $B_n$.
\e{\fr{\p}{\p y} u(x,l_2)=\sum_{n=1}^\infty \pi l_1^{-1} n B_n \sin(\pi l_1^{-1}n x)=0 \Rightarrow B=0}
The solution is 
\e{u(x,y)=\sum_{n=1}^\infty A_n \cosh(\pi l_1^{-1} n (\beta - y)) \sin(\pi l_1^{-1} n x)}
Solve for $A_n$ by evaluating $u(x,0)=g(x)$ and finding the Fourier series for $g(x)$.
\e{A_n = \fr{2}{l_1 \cosh(\pi l_1^{-1} l_2 n)} \int_0^{l_1} g(x) \sin(\pi l_1^{-1} n x)dx}
For this problem $g(x)=100$.
\e{A_n&=\fr{2}{\pi \cosh(n)}\int_0^\pi 100 \sin(nx)dx\\
&= \fr{-200}{\pi n \cosh(n)} ((-1)^n - 1)\\
&= \boxed{\falls{ \fr{400}{\pi n \cosh(n)} &\quad \textrm{$n$ odd}\\
0 &\quad \textrm{$n$ even}}}}
The solution is
\e{u(x,y)&=\sum_{n=1}^\infty A_n \cosh(\pi l_1^{-1} n (\beta - y)) \sin(\pi l_1^{-1} n x)\\
&= \boxed{\sum_{k=0}^\infty \fr{400}{\pi (2k+1)\cosh(2k+1)} \cosh((2k+1)(1-y))\sin((2k+1)x)}}
\section*{Problem 6}
This is a two dimensional Laplace equation with mixed Neumann and Robin boundary conditions. The solution is
\e{u(x,y)=\sum_{n=1}^\infty (A_n \cosh(\pi l_1^{-1} n y) + B_n \sinh (\pi l_1^{-1}n y))\sin(\pi l_1^{-1} n y)}
Solve for $B_n$ in terms of $A_n$.
\e{\fr{\p}{\p y} u(x,y)=\sum_{n=1}^\infty \pi l_1^{-1} n (A_n \sinh(\pi l_1^{-1} n y) + B_n \cosh (\pi l_1^{-1}y))\sin(\pi l_1^{-1} n y)}

\e{B_n=-2 l_1(\pi n)^{-1} A_n}

\e{\boxed{u(x,y)=\sum_{n=1}^\infty A_n (\cosh(\pi l_1^{-1} n y) - 2 l_1 (\pi n)^{-1} \sinh (\pi l_1^{-1}n y))\sin(\pi l_1^{-1} n y)}}

\e{\fr{\p}{\p y} u(x,l_2)=\sum_{n=1}^\infty A_n \pi l_1^{-1} n (\sinh(\pi l_1^{-1} l_2 n)- 2l_1 (\pi n)^{-1} \cosh(\pi l_1^{-1} l_2 n))=f(x)}
$A_n$ is

Solve this by cases. Case $f=0$. The solution for this case is $u_1$ with coefficients $\tilde{A_n}$, $\tilde{B_n}$. Solve for $\tilde{B_n}$.
\e{u_1(x,y)=\sum_{n=1}^\infty (\tilde{A_n}\cosh(\pi l_1^{-1} n (l_2-y)) +\tilde{B_n} \sinh(\pi l_1^{-1} n(l_2-y))\sin(\pi l_1^{-1} n x))}

\e{\tilde{B_n}=0}
Solve for $\tilde{A_n}$.
\e{u(x,0)+2\fr{\p}{\p y}u(x,0)=\sum_{n=1}^\infty \tilde{A_n}(\cosh(\pi l_1^{-1} l_2 n) -2\pi l_1^{-1} n \sinh(\pi l_1^{-1} l_2 n))\sin(\pi l_1^{-1} n x)=g(x)}
This is the Fourier sine series of $g(x)$. The coefficient is
\e{\boxed{\tilde{A_n}=\fr{2}{l_1 \cosh(\pi l_1^{-1}l_2 n)-2\pi n \sinh(\pi l_1^{-1} l_2 n)} \int_0^{l_1}g(x)\sin(\pi l_1^{-1} n x)dx}}
The second case is $g(x)=0$. The solution for this case is $u_2$ with coefficient $A_n$. This has been solved before.
\e{u_2(x,y)=\sum_{n=1}^\infty A_n(\cosh(\pi l_1^{-1} n y) - 2l_1 (\pi n)^{-1} \sinh (\pi l_1^{-1} n y))\sin(\pi l_1^{-1} n y)}
