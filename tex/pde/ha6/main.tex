\section*{Problem 8}
We'll take this problem step by step like in the notes. Find $h_n(t)$, then $T_n(0)$, then $T_n(t)$, then the general $U(t,x)$.
\e{\pt u=\ppx u}
\e{X_n(x)=\sin(\pnl x)\quad n=1,2,...}
\e{h(t,x)=\sin(3\pi x)=\sum_{n=1}^\infty h_n(t)\sin(\pnl x)=\sum_{n=1}^\infty h_n(t)\sin(n\pi x)}
\e{h_n(t)&=\fr{2}{l} \int_0^l h(t,x)\sin(\pnl)\\
&=2\int_0^1 \sin(3\pi x)\sin(n\pi x)dx\\
&=\begin{cases}1\quad &n=3 \\ 0 \quad &n\neq 0\end{cases}}
Next find $T_n(0)$.
\e{T'_n(t)+(\pnl)^2T_n(t)=h_n(t)}
\e{T_n(0)&=\fr{2}{l}\int_0^l f(x) \sin(\pnl x)dx\\
&= 2\int_0^1 \sin(\pi x)\sin(n\pi x)dx\\
&= \begin{cases} 0 \quad &n\neq 1 \\ 1 \quad &n=1\end{cases}}
Next, find $T_n(t)$. It will only be nonzero when $n=1$ or $n=3$. Use cases.
\e{T_n(t)&=e^{-(n\pi)^2 t}(T_n(0)+\int_0^1 e^{(n\pi)^2 s} h_n(s) ds)\\
&= \begin{cases} e^{-\pi^2 t} \quad &n=1 \\ e^{-(3\pi)^2 t} \int_0^1 e^{(3\pi)^2 s} ds \quad &n=3 \\ 0 \quad &\textrm{otherwise }\end{cases}\\
&= \begin{cases} e^{-\pi^2 t} \quad &n=1 \\ \fr{1}{(3\pi)^2} (1-e^{-(3\pi)^2} t)\quad &n=3 \\ 0\quad &\textrm{otherwise}\end{cases}}
After solving for $T_n(t)$ we can find the solution $U$. The solution $U$ will sum over $n$, and so will have two contributions: one term when $n=1$ and one term when $n=3$.
\e{U(t,x)&=\sum_{n=1}^\infty T_n(t) \sin(\pnl x)\\
&= T_1(t) \sin(\pi x)+T_3(t)\sin(3\pi x)\\
&= \boxed{e^{-\pi^2 t}\sin(\pi x)+\fr{1}{(3\pi)^2}(1-e^{-(3\pi)^2} t)\sin(3\pi x)}}
